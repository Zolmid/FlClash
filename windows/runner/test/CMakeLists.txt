cmake_minimum_required(VERSION 3.14)
project(autostart_cleanup_tests LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(autostart_cleanup_tests autostart_cleanup_test.cpp)
target_include_directories(autostart_cleanup_tests PRIVATE ..)
if(WIN32)
  target_sources(autostart_cleanup_tests PRIVATE ../autostart_cleanup.cpp)
  target_link_libraries(autostart_cleanup_tests PRIVATE taskschd ole32 oleaut32 advapi32 bcrypt user32)
  target_compile_definitions(autostart_cleanup_tests PRIVATE UNICODE _UNICODE NOMINMAX)
endif()
if(MSVC)
  target_compile_options(autostart_cleanup_tests PRIVATE /W4 /WX)
else()
  target_compile_options(autostart_cleanup_tests PRIVATE -Wall -Wextra -Werror)
endif()
foreach(case owned other_user other_install source command shape principal arguments)
  add_test(NAME autostart_${case} COMMAND autostart_cleanup_tests ${case})
endforeach()
if(WIN32)
  add_test(NAME autostart_native_hash COMMAND autostart_cleanup_tests native_hash)
endif()
